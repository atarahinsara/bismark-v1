     1→     1→     1→/**
     2→     2→     2→ * Notification Service — Core Dispatcher + Retry Engine (LAW-55/56/57)
     3→     3→     3→ * =====================================================================
     4→     4→     4→ *
     5→     5→     5→ * The Notification Context's central service. Responsibilities:
     6→     6→     6→ *
     7→     7→     7→ *   dispatch(input)         — render template, dedupe by idempotencyKey,
     8→     8→     8→ *                             create Notification + Queue item, publish
     9→     9→     9→ *                             outbox events (LAW-55 + LAW-57).
    10→    10→    10→ *
    11→    11→    11→ *   processQueueItem(...)   — retry engine. Locks a queue item, calls the
    12→    12→    12→ *                             channel provider, records the delivery, and
    13→    13→    13→ *                             on failure schedules a backoff retry or
    14→    14→    14→ *                             moves the item to the DLQ (LAW-57).
    15→    15→    15→ *
    16→    16→    16→ *   cancel(...)             — terminal cancellation (clears queue items).
    17→    17→    17→ *
    18→    18→    18→ *   retry(...)              — manual retry of a DLQ'd notification.
    19→    19→    19→ *
    20→    20→    20→ *   list / getById / getStats — read-side helpers for the admin UI.
    21→    21→    21→ *
    22→    22→    22→ * Architecture laws:
    23→    23→    23→ *   LAW-08: Every state change publishes an outbox event
    24→    24→    24→ *   LAW-55: Notifications Must Be Template-Based (versioned, language-aware)
    25→    25→    25→ *   LAW-56: Notification Delivery Must Be Channel-Agnostic
    26→    26→    26→ *   LAW-57: Notification Delivery Must Be Retryable And Idempotent
    27→    27→    27→ */
    28→    28→    28→
    29→    29→    29→import { db } from '@/lib/db'
    30→    30→    30→import { getTenantId } from '@/lib/api-helpers'
    31→    31→    31→import {
    32→    32→    32→  UnitOfWork,
    33→    33→    33→  NotFoundException,
    34→    34→    34→  BusinessException,
    35→    35→    35→} from '@/lib/shared'
    36→    36→    36→import { renderTemplate } from './template-engine'
    37→    37→    37→import { getProvider } from './providers'
    38→    38→    38→import type {
    39→    39→    39→  Channel,
    40→    40→    40→  DispatchInput,
    41→    41→    41→  DispatchResult,
    42→    42→    42→  NotificationStatus,
    43→    43→    43→} from './types'
    44→    44→    44→
    45→    45→    45→// ============================================================
    46→    46→    46→// Backoff schedule (LAW-57)
    47→    47→    47→// ============================================================
    48→    48→    48→
    49→    49→    49→/**
    50→    50→    50→ * Retry backoff in seconds, indexed by `attempt - 1`.
    51→    51→    51→ *
    52→    52→    52→ *   attempt 1 fails → wait 60s
    53→    53→    53→ *   attempt 2 fails → wait 300s   (5 min)
    54→    54→    54→ *   attempt 3 fails → wait 1800s  (30 min)
    55→    55→    55→ *   attempt 4 fails → wait 7200s  (2 hours)
    56→    56→    56→ *   attempt 5 fails → DLQ (terminal)
    57→    57→    57→ *
    58→    58→    58→ * maxAttempts defaults to 5 (see dispatch step 6).
    59→    59→    59→ */
    60→    60→    60→const BACKOFF_SCHEDULE_SECONDS = [60, 300, 1800, 7200]
    61→    61→    61→
    62→    62→    62→/** Maximum attempts per notification (LAW-57). */
    63→    63→    63→const DEFAULT_MAX_ATTEMPTS = 5
    64→    64→    64→
    65→    65→    65→// ============================================================
    66→    66→    66→// Service
    67→    67→    67→// ============================================================
    68→    68→    68→
    69→    69→    69→export class NotificationService {
    70→    70→    70→  // ----------------------------------------------------------
    71→    71→    71→  // dispatch (LAW-55 + LAW-57)
    72→    72→    72→  // ----------------------------------------------------------
    73→    73→    73→
    74→    74→    74→  /**
    75→    75→    75→   * Render the named template, dedupe by `input.idempotencyKey`, create a
    76→    76→    76→   * Notification + Queue item inside a transaction, and publish outbox
    77→    77→    77→   * events `notification.created` and `notification.queued`.
    78→    78→    78→   *
    79→    79→    79→   * Idempotent: a second call with the same idempotencyKey returns the
    80→    80→    80→   * existing notification with `created: false`.
    81→    81→    81→   */
    82→    82→    82→  async dispatch(input: DispatchInput): Promise<DispatchResult> {
    83→    83→    83→    const tenantId = await getTenantId()
    84→    84→    84→
    85→    85→    85→    // 1. Idempotency pre-check (LAW-57)
    86→    86→    86→    const existing = await db.notification.findUnique({
    87→    87→    87→      where: { tenantId_idempotencyKey: { tenantId, idempotencyKey: input.idempotencyKey } },
    88→    88→    88→    })
    89→    89→    89→    if (existing) {
    90→    90→    90→      return {
    91→    91→    91→        notificationId: existing.id,
    92→    92→    92→        status: existing.status as NotificationStatus,
    93→    93→    93→        created: false,
    94→    94→    94→        message: 'idempotent_hit',
    95→    95→    95→      }
    96→    96→    96→    }
    97→    97→    97→
    98→    98→    98→    // 2. Find the published template (LAW-55: versioned, language-aware)
    99→    99→    99→    const template = await this.findTemplate(tenantId, input.templateCode, input.language)
   100→   100→   100→    if (!template) {
   101→   101→   101→      throw new NotFoundException('NOTIFICATION_TEMPLATE', input.templateCode)
   102→   102→   102→    }
   103→   103→   103→
   104→   104→   104→    // 3. Resolve channel — caller override wins, else template's channel
   105→   105→   105→    const channel: Channel = (input.channel ?? (template.channel as Channel)) as Channel
   106→   106→   106→
   107→   107→   107→    // 4. Render (LAW-55: deterministic)
   108→   108→   108→    const rendered = renderTemplate({
   109→   109→   109→      subjectTemplate: template.subjectTemplate,
   110→   110→   110→      bodyTemplate: template.bodyTemplate,
   111→   111→   111→      variables: input.variables,
   112→   112→   112→    })
   113→   113→   113→    if (rendered.warnings.length > 0) {
   114→   114→   114→      console.warn(
   115→   115→   115→        `[notification] template "${template.code}" v${template.version} produced ` +
   116→   116→   116→          `${rendered.warnings.length} warning(s): ${rendered.warnings.join('; ')}`,
   117→   117→   117→      )
   118→   118→   118→    }
   119→   119→   119→
   120→   120→   120→    // 5. Persist inside a UoW (LAW-08: outbox events in same transaction)
   121→   121→   121→    const now = new Date()
   122→   122→   122→    const priority = input.priority ?? 100
   123→   123→   123→    const language = input.language ?? template.language
   124→   124→   124→
   125→   125→   125→    try {
   126→   126→   126→      const notification = await UnitOfWork.execute(async (uow) => {
   127→   127→   127→        // (a) Create Notification with status=pending
   128→   128→   128→        const notif = await uow.tx.notification.create({
   129→   129→   129→          data: {
   130→   130→   130→            tenantId,
   131→   131→   131→            templateId: template.id,
   132→   132→   132→            templateCode: template.code,
   133→   133→   133→            templateVersion: template.version,
   134→   134→   134→            language,
   135→   135→   135→            recipientId: input.recipientId ?? null,
   136→   136→   136→            recipientName: input.recipientName ?? null,
   137→   137→   137→            recipientAddress: input.recipientAddress,
   138→   138→   138→            channel,
   139→   139→   139→            status: 'pending',
   140→   140→   140→            payload: input.variables as any,
   141→   141→   141→            renderedSubject: rendered.subject,
   142→   142→   142→            renderedBody: rendered.body,
   143→   143→   143→            idempotencyKey: input.idempotencyKey,
   144→   144→   144→          },
   145→   145→   145→        })
   146→   146→   146→
   147→   147→   147→        // (b) Create Queue item (LAW-57)
   148→   148→   148→        await uow.tx.notificationQueue.create({
   149→   149→   149→          data: {
   150→   150→   150→            tenantId,
   151→   151→   151→            notificationId: notif.id,
   152→   152→   152→            priority,
   153→   153→   153→            attempt: 0,
   154→   154→   154→            maxAttempts: DEFAULT_MAX_ATTEMPTS,
   155→   155→   155→            nextRetryAt: now,
   156→   156→   156→            inDeadLetter: false,
   157→   157→   157→          },
   158→   158→   158→        })
   159→   159→   159→
   160→   160→   160→        // (c) Outbox: notification.created
   161→   161→   161→        await uow.outbox.append({
   162→   162→   162→          tenantId,
   163→   163→   163→          aggregateType: 'Notification',
   164→   164→   164→          aggregateId: notif.id,
   165→   165→   165→          eventType: 'notification.created',
   166→   166→   166→          eventVersion: '1.0',
   167→   167→   167→          payload: {
   168→   168→   168→            notificationId: notif.id,
   169→   169→   169→            templateCode: template.code,
   170→   170→   170→            templateVersion: template.version,
   171→   171→   171→            language,
   172→   172→   172→            channel,
   173→   173→   173→            recipientId: input.recipientId ?? null,
   174→   174→   174→            recipientAddress: input.recipientAddress,
   175→   175→   175→            triggeredByEvent: input.triggeredByEvent ?? null,
   176→   176→   176→            status: 'pending',
   177→   177→   177→          },
   178→   178→   178→          actorId: null,
   179→   179→   179→        })
   180→   180→   180→
   181→   181→   181→        // (d) Transition to queued
   182→   182→   182→        const updated = await uow.tx.notification.update({
   183→   183→   183→          where: { id: notif.id },
   184→   184→   184→          data: { status: 'queued', queuedAt: now },
   185→   185→   185→        })
   186→   186→   186→
   187→   187→   187→        // (e) Outbox: notification.queued
   188→   188→   188→        await uow.outbox.append({
   189→   189→   189→          tenantId,
   190→   190→   190→          aggregateType: 'Notification',
   191→   191→   191→          aggregateId: notif.id,
   192→   192→   192→          eventType: 'notification.queued',
   193→   193→   193→          eventVersion: '1.0',
   194→   194→   194→          payload: {
   195→   195→   195→            notificationId: notif.id,
   196→   196→   196→            priority,
   197→   197→   197→            maxAttempts: DEFAULT_MAX_ATTEMPTS,
   198→   198→   198→            queuedAt: now.toISOString(),
   199→   199→   199→          },
   200→   200→   200→          actorId: null,
   201→   201→   201→        })
   202→   202→   202→
   203→   203→   203→        return updated
   204→   204→   204→      })
   205→   205→   205→
   206→   206→   206→      console.log(
   207→   207→   207→        `[notification] dispatched ${template.code} v${template.version} ` +
   208→   208→   208→          `(${channel}) → notification ${notification.id} [queued]`,
   209→   209→   209→      )
   210→   210→   210→
   211→   211→   211→      return {
   212→   212→   212→        notificationId: notification.id,
   213→   213→   213→        status: 'queued',
   214→   214→   214→        created: true,
   215→   215→   215→      }
   216→   216→   216→    } catch (err: any) {
   217→   217→   217→      // P2002: unique constraint violation — race condition (another worker
   218→   218→   218→      // created the same idempotencyKey concurrently). Refetch and return
   219→   219→   219→      // as idempotent hit (LAW-57).
   220→   220→   220→      if (err?.code === 'P2002') {
   221→   221→   221→        const raced = await db.notification.findUnique({
   222→   222→   222→          where: {
   223→   223→   223→            tenantId_idempotencyKey: { tenantId, idempotencyKey: input.idempotencyKey },
   224→   224→   224→          },
   225→   225→   225→        })
   226→   226→   226→        if (raced) {
   227→   227→   227→          console.log(
   228→   228→   228→            `[notification] race-condition idempotent hit for key=${input.idempotencyKey}`,
   229→   229→   229→          )
   230→   230→   230→          return {
   231→   231→   231→            notificationId: raced.id,
   232→   232→   232→            status: raced.status as NotificationStatus,
   233→   233→   233→            created: false,
   234→   234→   234→            message: 'idempotent_hit_after_race',
   235→   235→   235→          }
   236→   236→   236→        }
   237→   237→   237→      }
   238→   238→   238→      throw err
   239→   239→   239→    }
   240→   240→   240→  }
   241→   241→   241→
   242→   242→   242→  // ----------------------------------------------------------
   243→   243→   243→  // processQueueItem — retry engine (LAW-57)
   244→   244→   244→  // ----------------------------------------------------------
   245→   245→   245→
   246→   246→   246→  /**
   247→   247→   247→   * Process a single queue item end-to-end. Called by the queue worker
   248→   248→   248→   * (cron / manual). Behaviour:
   249→   249→   249→   *
   250→   250→   250→   *   - Skip if already locked or in DLQ
   251→   251→   251→   *   - Skip if notification is in a terminal state (sent / cancelled)
   252→   252→   252→   *   - DLQ immediately if next attempt would exceed maxAttempts
   253→   253→   253→   *   - Otherwise: pick provider, create delivery, send
   254→   254→   254→   *     - Success → mark sent (terminal)
   255→   255→   255→   *     - Failure → schedule backoff retry, or DLQ on final attempt
   256→   256→   256→   */
   257→   257→   257→  async processQueueItem(
   258→   258→   258→    queueItemId: string,
   259→   259→   259→    workerId: string,
   260→   260→   260→  ): Promise<{
   261→   261→   261→    status: 'sent' | 'failed' | 'retrying' | 'dead_lettered' | 'skipped'
   262→   262→   262→    message?: string
   263→   263→   263→  }> {
   264→   264→   264→    const now = new Date()
   265→   265→   265→
   266→   266→   266→    // 1. Conditional lock (LAW-57: only if not DLQ'd and not locked)
   267→   267→   267→    const lockResult = await db.notificationQueue.updateMany({
   268→   268→   268→      where: { id: queueItemId, inDeadLetter: false, lockedBy: null },
   269→   269→   269→      data: { lockedBy: workerId, lockedAt: now },
   270→   270→   270→    })
   271→   271→   271→    if (lockResult.count === 0) {
   272→   272→   272→      return { status: 'skipped', message: 'already_locked_or_in_dlq' }
   273→   273→   273→    }
   274→   274→   274→
   275→   275→   275→    // 2. Load queue item
   276→   276→   276→    const queueItem = await db.notificationQueue.findUnique({
   277→   277→   277→      where: { id: queueItemId },
   278→   278→   278→    })
   279→   279→   279→    if (!queueItem) {
   280→   280→   280→      return { status: 'skipped', message: 'queue_item_not_found' }
   281→   281→   281→    }
   282→   282→   282→
   283→   283→   283→    // 3. Load notification; skip + unlock if terminal
   284→   284→   284→    const notification = await db.notification.findUnique({
   285→   285→   285→      where: { id: queueItem.notificationId },
   286→   286→   286→    })
   287→   287→   287→    if (!notification) {
   288→   288→   288→      await this.unlockQueueItem(queueItemId)
   289→   289→   289→      return { status: 'skipped', message: 'notification_not_found' }
   290→   290→   290→    }
   291→   291→   291→    if (notification.status === 'sent' || notification.status === 'cancelled') {
   292→   292→   292→      await this.unlockQueueItem(queueItemId)
   293→   293→   293→      return {
   294→   294→   294→        status: 'skipped',
   295→   295→   295→        message: `notification_${notification.status}`,
   296→   296→   296→      }
   297→   297→   297→    }
   298→   298→   298→
   299→   299→   299→    // 4. Compute next attempt; pre-check DLQ (defensive — should rarely fire
   300→   300→   300→    //    because the queue item is normally already in DLQ at this point)
   301→   301→   301→    const newAttempt = queueItem.attempt + 1
   302→   302→   302→    if (newAttempt > queueItem.maxAttempts) {
   303→   303→   303→      await UnitOfWork.execute(async (uow) => {
   304→   304→   304→        await uow.tx.notificationQueue.update({
   305→   305→   305→          where: { id: queueItemId },
   306→   306→   306→          data: {
   307→   307→   307→            inDeadLetter: true,
   308→   308→   308→            deadLetterAt: now,
   309→   309→   309→            deadLetterReason: 'max_attempts_exceeded',
   310→   310→   310→          },
   311→   311→   311→        })
   312→   312→   312→        await uow.tx.notification.update({
   313→   313→   313→          where: { id: notification.id },
   314→   314→   314→          data: { status: 'failed', failedAt: now },
   315→   315→   315→        })
   316→   316→   316→        await uow.outbox.append({
   317→   317→   317→          tenantId: notification.tenantId,
   318→   318→   318→          aggregateType: 'Notification',
   319→   319→   319→          aggregateId: notification.id,
   320→   320→   320→          eventType: 'notification.failed',
   321→   321→   321→          eventVersion: '1.0',
   322→   322→   322→          payload: {
   323→   323→   323→            notificationId: notification.id,
   324→   324→   324→            movedToDLQ: true,
   325→   325→   325→            reason: 'max_attempts_exceeded',
   326→   326→   326→            attempts: queueItem.attempt,
   327→   327→   327→            maxAttempts: queueItem.maxAttempts,
   328→   328→   328→          },
   329→   329→   329→          actorId: null,
   330→   330→   330→        })
   331→   331→   331→      })
   332→   332→   332→      console.error(
   333→   333→   333→        `[notification] DLQ ${notification.id} — max_attempts_exceeded ` +
   334→   334→   334→          `(attempts=${queueItem.attempt}, max=${queueItem.maxAttempts})`,
   335→   335→   335→      )
   336→   336→   336→      return { status: 'dead_lettered', message: 'max_attempts_exceeded' }
   337→   337→   337→    }
   338→   338→   338→
   339→   339→   339→    // 5. Pick provider (LAW-56)
   340→   340→   340→    const provider = getProvider(notification.channel as Channel)
   341→   341→   341→
   342→   342→   342→    // 6. Create NotificationDelivery row (attempt, sending)
   343→   343→   343→    const delivery = await db.notificationDelivery.create({
   344→   344→   344→      data: {
   345→   345→   345→        tenantId: notification.tenantId,
   346→   346→   346→        notificationId: notification.id,
   347→   347→   347→        provider: provider.name,
   348→   348→   348→        attempt: newAttempt,
   349→   349→   349→        status: 'sending',
   350→   350→   350→        durationMs: 0,
   351→   351→   351→        createdAt: now,
   352→   352→   352→      },
   353→   353→   353→    })
   354→   354→   354→
   355→   355→   355→    // 7. Mark notification as sending
   356→   356→   356→    await db.notification.update({
   357→   357→   357→      where: { id: notification.id },
   358→   358→   358→      data: { status: 'sending' },
   359→   359→   359→    })
   360→   360→   360→
   361→   361→   361→    // 8. Send via provider (never throws — returns failure result)
   362→   362→   362→    const sendResult = await provider.send({
   363→   363→   363→      to: notification.recipientAddress,
   364→   364→   364→      subject: notification.renderedSubject,
   365→   365→   365→      body: notification.renderedBody,
   366→   366→   366→      variables: (notification.payload as Record<string, any>) ?? {},
   367→   367→   367→    })
   368→   368→   368→
   369→   369→   369→    // 9. Success path
   370→   370→   370→    if (sendResult.success) {
   371→   371→   371→      await UnitOfWork.execute(async (uow) => {
   372→   372→   372→        await uow.tx.notificationDelivery.update({
   373→   373→   373→          where: { id: delivery.id },
   374→   374→   374→          data: {
   375→   375→   375→            status: 'sent',
   376→   376→   376→            response: (sendResult.providerResponse as any) ?? null,
   377→   377→   377→            durationMs: sendResult.durationMs,
   378→   378→   378→          },
   379→   379→   379→        })
   380→   380→   380→        await uow.tx.notification.update({
   381→   381→   381→          where: { id: notification.id },
   382→   382→   382→          data: {
   383→   383→   383→            status: 'sent',
   384→   384→   384→            messageId: sendResult.messageId ?? null,
   385→   385→   385→            sentAt: now,
   386→   386→   386→            // Clear any previous error info from earlier failed attempts
   387→   387→   387→            errorCode: null,
   388→   388→   388→            errorMessage: null,
   389→   389→   389→          },
   390→   390→   390→        })
   391→   391→   391→        await uow.tx.notificationQueue.update({
   392→   392→   392→          where: { id: queueItemId },
   393→   393→   393→          data: { attempt: newAttempt },
   394→   394→   394→          // lockedBy/lockedAt left set — terminal state, won't be picked up again
   395→   395→   395→        })
   396→   396→   396→        await uow.outbox.append({
   397→   397→   397→          tenantId: notification.tenantId,
   398→   398→   398→          aggregateType: 'Notification',
   399→   399→   399→          aggregateId: notification.id,
   400→   400→   400→          eventType: 'notification.sent',
   401→   401→   401→          eventVersion: '1.0',
   402→   402→   402→          payload: {
   403→   403→   403→            notificationId: notification.id,
   404→   404→   404→            provider: provider.name,
   405→   405→   405→            attempt: newAttempt,
   406→   406→   406→            messageId: sendResult.messageId ?? null,
   407→   407→   407→            durationMs: sendResult.durationMs,
   408→   408→   408→          },
   409→   409→   409→          actorId: null,
   410→   410→   410→        })
   411→   411→   411→      })
   412→   412→   412→      console.log(
   413→   413→   413→        `[notification] sent ${notification.id} via ${provider.name} ` +
   414→   414→   414→          `(attempt ${newAttempt}/${queueItem.maxAttempts}, ${sendResult.durationMs}ms)`,
   415→   415→   415→      )
   416→   416→   416→      return { status: 'sent' }
   417→   417→   417→    }
   418→   418→   418→
   419→   419→   419→    // 10. Failure path
   420→   420→   420→    console.error(
   421→   421→   421→      `[notification] send failed for ${notification.id} via ${provider.name} ` +
   422→   422→   422→        `(attempt ${newAttempt}/${queueItem.maxAttempts}): ${sendResult.errorMessage}`,
   423→   423→   423→    )
   424→   424→   424→
   425→   425→   425→    const isFinalAttempt = newAttempt >= queueItem.maxAttempts
   426→   426→   426→
   427→   427→   427→    if (isFinalAttempt) {
   428→   428→   428→      // 10a. Move to DLQ
   429→   429→   429→      await UnitOfWork.execute(async (uow) => {
   430→   430→   430→        await uow.tx.notificationDelivery.update({
   431→   431→   431→          where: { id: delivery.id },
   432→   432→   432→          data: {
   433→   433→   433→            status: 'failed',
   434→   434→   434→            errorMessage: sendResult.errorMessage ?? null,
   435→   435→   435→            durationMs: sendResult.durationMs,
   436→   436→   436→          },
   437→   437→   437→        })
   438→   438→   438→        await uow.tx.notificationQueue.update({
   439→   439→   439→          where: { id: queueItemId },
   440→   440→   440→          data: {
   441→   441→   441→            attempt: newAttempt,
   442→   442→   442→            inDeadLetter: true,
   443→   443→   443→            deadLetterAt: now,
   444→   444→   444→            deadLetterReason: sendResult.errorMessage ?? 'unknown_error',
   445→   445→   445→          },
   446→   446→   446→        })
   447→   447→   447→        await uow.tx.notification.update({
   448→   448→   448→          where: { id: notification.id },
   449→   449→   449→          data: {
   450→   450→   450→            status: 'failed',
   451→   451→   451→            failedAt: now,
   452→   452→   452→            errorCode: 'DELIVERY_FAILED',
   453→   453→   453→            errorMessage: sendResult.errorMessage ?? 'unknown_error',
   454→   454→   454→          },
   455→   455→   455→        })
   456→   456→   456→        await uow.outbox.append({
   457→   457→   457→          tenantId: notification.tenantId,
   458→   458→   458→          aggregateType: 'Notification',
   459→   459→   459→          aggregateId: notification.id,
   460→   460→   460→          eventType: 'notification.failed',
   461→   461→   461→          eventVersion: '1.0',
   462→   462→   462→          payload: {
   463→   463→   463→            notificationId: notification.id,
   464→   464→   464→            movedToDLQ: true,
   465→   465→   465→            attempt: newAttempt,
   466→   466→   466→            maxAttempts: queueItem.maxAttempts,
   467→   467→   467→            provider: provider.name,
   468→   468→   468→            errorMessage: sendResult.errorMessage ?? null,
   469→   469→   469→          },
   470→   470→   470→          actorId: null,
   471→   471→   471→        })
   472→   472→   472→      })
   473→   473→   473→      return { status: 'dead_lettered', message: sendResult.errorMessage }
   474→   474→   474→    }
   475→   475→   475→
   476→   476→   476→    // 10b. Schedule a backoff retry
   477→   477→   477→    const backoffIndex = Math.min(newAttempt - 1, BACKOFF_SCHEDULE_SECONDS.length - 1)
   478→   478→   478→    const backoffSeconds = BACKOFF_SCHEDULE_SECONDS[backoffIndex]
   479→   479→   479→    const nextRetryAt = new Date(now.getTime() + backoffSeconds * 1000)
   480→   480→   480→
   481→   481→   481→    await UnitOfWork.execute(async (uow) => {
   482→   482→   482→      await uow.tx.notificationDelivery.update({
   483→   483→   483→        where: { id: delivery.id },
   484→   484→   484→        data: {
   485→   485→   485→          status: 'failed',
   486→   486→   486→          errorMessage: sendResult.errorMessage ?? null,
   487→   487→   487→          durationMs: sendResult.durationMs,
   488→   488→   488→        },
   489→   489→   489→      })
   490→   490→   490→      await uow.tx.notificationQueue.update({
   491→   491→   491→        where: { id: queueItemId },
   492→   492→   492→        data: {
   493→   493→   493→          attempt: newAttempt,
   494→   494→   494→          nextRetryAt,
   495→   495→   495→          // Unlock so the worker can pick it up at nextRetryAt
   496→   496→   496→          lockedBy: null,
   497→   497→   497→          lockedAt: null,
   498→   498→   498→        },
   499→   499→   499→      })
   500→   500→   500→      await uow.tx.notification.update({
   501→   501→   501→        where: { id: notification.id },
   502→   502→   502→        data: {
   503→   503→   503→          status: 'retrying',
   504→   504→   504→          errorCode: 'DELIVERY_FAILED',
   505→   505→   505→          errorMessage: sendResult.errorMessage ?? null,
   506→   506→   506→        },
   507→   507→   507→      })
   508→   508→   508→      await uow.outbox.append({
   509→   509→   509→        tenantId: notification.tenantId,
   510→   510→   510→        aggregateType: 'Notification',
   511→   511→   511→        aggregateId: notification.id,
   512→   512→   512→        eventType: 'notification.retrying',
   513→   513→   513→        eventVersion: '1.0',
   514→   514→   514→        payload: {
   515→   515→   515→          notificationId: notification.id,
   516→   516→   516→          attempt: newAttempt,
   517→   517→   517→          maxAttempts: queueItem.maxAttempts,
   518→   518→   518→          nextRetryAt: nextRetryAt.toISOString(),
   519→   519→   519→          backoffSeconds,
   520→   520→   520→          provider: provider.name,
   521→   521→   521→          errorMessage: sendResult.errorMessage ?? null,
   522→   522→   522→        },
   523→   523→   523→        actorId: null,
   524→   524→   524→      })
   525→   525→   525→    })
   526→   526→   526→
   527→   527→   527→    console.log(
   528→   528→   528→      `[notification] retrying ${notification.id} in ${backoffSeconds}s ` +
   529→   529→   529→        `(attempt ${newAttempt}/${queueItem.maxAttempts})`,
   530→   530→   530→    )
   531→   531→   531→    return { status: 'retrying', message: `next_retry_in_${backoffSeconds}s` }
   532→   532→   532→  }

   736→   736→   736→  async getStats(tenantId: string): Promise<{

   746→   746→   746→  }> {

   748→   748→   748→    const todayStart = new Date(
   749→   749→   749→      now.getFullYear(),

   827→   827→   827→
   828→   828→   828→  /**
   829→   829→   829→   * Find the latest published template for `code` with language fallback.
   830→   830→   830→   * Order: input.language (if given) → 'fa' → any.
   831→   831→   831→   *
   832→   832→   832→   * Filters (LAW-55):
   833→   833→   833→   *   - status = 'published'
   834→   834→   834→   *   - effectiveFrom <= now
   835→   835→   835→   *   - effectiveTo IS NULL OR effectiveTo >= now
   836→   836→   836→   *   - deletedAt IS NULL
   837→   837→   837→   * Sort: version DESC (latest published version wins).
   838→   838→   838→   */
   839→   839→   839→  private async findTemplate(
   840→   840→   840→    tenantId: string,
   841→   841→   841→    code: string,
   842→   842→   842→    language?: string,
   843→   843→   843→  ): Promise<any | null> {
   844→   844→   844→    const now = new Date()
   845→   845→   845→    const baseWhere = {
   846→   846→   846→      tenantId,
   847→   847→   847→      code,
   848→   848→   848→      status: 'published',
   849→   849→   849→      deletedAt: null,
   850→   850→   850→      effectiveFrom: { lte: now },
   851→   851→   851→      OR: [{ effectiveTo: null }, { effectiveTo: { gte: now } }],
   852→   852→   852→    }
   853→   853→   853→
   854→   854→   854→    // Try preferred language, then 'fa' as a fallback
   855→   855→   855→    const preferredLanguages = language ? [language, 'fa'] : ['fa']
   856→   856→   856→    for (const lang of preferredLanguages) {
   857→   857→   857→      const tpl = await db.notificationTemplate.findFirst({
