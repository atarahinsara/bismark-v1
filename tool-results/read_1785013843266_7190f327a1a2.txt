    86→function timeAgo(date: string | null): string {
    87→  if (!date) return '—'
    88→  const diff = Date.now() - new Date(date).getTime()
    89→  const minutes = Math.floor(diff / 60000)
    90→  if (minutes < 1) return 'همین الان'
    91→  if (minutes < 60) return `${minutes} دقیقه پیش`
    92→  const hours = Math.floor(minutes / 60)
    93→  if (hours < 24) return `${hours} ساعت پیش`
    94→  const days = Math.floor(hours / 24)
    95→  return `${days} روز پیش`
    96→}
    97→
    98→function formatDate(date: string): string {
    99→  return new Date(date).toLocaleDateString('fa-IR', { year: 'numeric', month: 'long', day: 'numeric' })
   100→}
