    20→model IdempotencyKey {
    21→  id              String   @id @default(cuid())
    22→  tenantId        String
    23→  key             String // the Idempotency-Key header value (UUID)
    24→  endpoint        String // e.g., "POST /api/v1/inventory-transactions"
    25→  requestHash     String // SHA-256 of request body
    26→  responseBody    String // cached response (JSON)
    27→  responseStatus  Int // cached HTTP status
    28→  responseHeaders Json? // cached headers
    29→  expiresAt       DateTime // 24h after creation
    30→  createdAt       DateTime @default(now())
    31→
    32→  @@unique([tenantId, key])
    33→  @@index([expiresAt])
    34→  @@map("idempotency_keys")
    35→}
